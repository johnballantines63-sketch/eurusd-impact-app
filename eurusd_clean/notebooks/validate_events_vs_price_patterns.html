🧱 CELLULE 4 — Chargement du consensus (estimate / prev)¶

⚠️ Important
economic_events utilise:

  • datetime_utc (pas ts_utc)
  • event_name (pas event)
  • forecast (pas estimate) — mais on le renomme en estimate pour uniformiser
  • previous (pas prev) — mais on le renomme en prev pour uniformiser

Source : economic_events

In [4]:
def load_consensus(date):
    """
    Charge le consensus depuis economic_events.
    Note: economic_events utilise datetime_utc (pas ts_utc), 
    event_name (pas event), forecast (pas estimate), previous (pas prev).
    """
    df = conn.execute("""
        SELECT
            datetime_utc,
            country,
            event_name,
            forecast,
            previous
        FROM economic_events
        WHERE DATE(datetime_utc) = CAST(? AS DATE)
    """, [date]).df()
    
    # Renommer pour uniformiser avec le reste du notebook
    if not df.empty:
        df = df.rename(columns={
            "datetime_utc": "ts_utc",
            "event_name": "event",
            "forecast": "estimate",
            "previous": "prev"
        })
    
    return df

def normalize(text):
    return (
        text.astype(str)
            .str.lower()
            .str.replace(r"[^a-z0-9 ]", "", regex=True)
            .str.replace(r"\s+", " ", regex=True)

    )

def merge_events_and_consensus(events_df, cons_df):
    """
    Matching robuste : ts_utc + country (canonique) puis fallback texte.
    
    Note: cons_df doit avoir été préalablement transformé par load_consensus()
    qui renomme datetime_utc→ts_utc, event_name→event, forecast→estimate, previous→prev.
    """
    ev = events_df.copy()
    co = cons_df.copy()

    # Convertir ts_utc en datetime
    ev["ts_utc"] = pd.to_datetime(ev["ts_utc"])
    co["ts_utc"] = pd.to_datetime(co["ts_utc"])

    # Join canonique : ts_utc + country (prioritaire)
    merged = ev.merge(
        co,
        on=["ts_utc", "country"],
        how="left",
        suffixes=("_event", "_cons")
    )

    # Fallback: pour les lignes sans estimate, essayer matching texte
    missing = merged["estimate"].isna()
    if missing.any():
        ev2 = merged.loc[missing, ["ts_utc", "ts_local", "country", "event_title", "actual"]].copy()
        ev2["key_ev"] = normalize(ev2["event_title"])
        
        co2 = cons_df.copy()
        co2["key_co"] = normalize(co2["event"])
        co2["ts_utc"] = pd.to_datetime(co2["ts_utc"])

        # Fallback texte sur country + key
        fb = ev2.merge(
            co2,
            left_on=["country", "key_ev"],
            right_on=["country", "key_co"],
            how="left"
        )
        
        # Réinjecte estimate/prev si trouvé
        merged.loc[missing, "estimate"] = fb["estimate"].values
        merged.loc[missing, "prev"] = fb["prev"].values

    return merged

def load_prices_full_day(date):
    """
    Charge les prix pour toute la journée (00:00 à 23:59:59).
    Note: prices_finnhub_m5 utilise 'datetime' (pas ts_utc).
    """
    df = conn.execute("""
        SELECT datetime, close
        FROM prices_finnhub_m5
        WHERE datetime >= CAST(? AS TIMESTAMP)
          AND datetime <  CAST(? AS TIMESTAMP) + INTERVAL '1 day'
        ORDER BY datetime
    """, [date + " 00:00:00", date + " 00:00:00"]).df()
    
    # Renommer datetime en ts_utc pour uniformiser
    if not df.empty:
        df = df.rename(columns={"datetime": "ts_utc"})
    
    return df


def load_prices_around_events(date, events_df, before_hours=2, after_hours=6):
    """
    Charge les prix autour des événements du jour (fenêtre centrée).
    Note: prices_finnhub_m5 utilise 'datetime' (pas ts_utc).
    """
    if events_df.empty:
        return load_prices_full_day(date)

    ev_copy = events_df.copy()
    ev_copy["ts_utc"] = pd.to_datetime(ev_copy["ts_utc"])
    
    t0 = ev_copy["ts_utc"].min()
    t1 = ev_copy["ts_utc"].max()

    start = (t0 - pd.Timedelta(hours=before_hours)).strftime("%Y-%m-%d %H:%M:%S")
    end   = (t1 + pd.Timedelta(hours=after_hours)).strftime("%Y-%m-%d %H:%M:%S")

    df = conn.execute("""
        SELECT datetime, close
        FROM prices_finnhub_m5
        WHERE datetime >= CAST(? AS TIMESTAMP)
          AND datetime <= CAST(? AS TIMESTAMP)
        ORDER BY datetime
    """, [start, end]).df()
    
    # Renommer datetime en ts_utc pour uniformiser
    if not df.empty:
        df = df.rename(columns={"datetime": "ts_utc"})
    
    return df
